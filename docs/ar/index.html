class AuthBloc extends Bloc<AuthEvent, AuthState> {
  final SignInUseCase _signIn;

  AuthBloc({required SignInUseCase signIn})
      : _signIn = signIn,
        super(const AuthState.initial()) {
    on<SignInSubmitted>(_onSignInSubmitted);
  }

  Future<void> _onSignInSubmitted(
    SignInSubmitted event,
    Emitter<AuthState> emit,
  ) async {
    emit(const AuthState.loading());
    final result = await _signIn(event.params);
    result.fold(
      (failure) => emit(AuthState.error(failure.message)),
      (user) => emit(AuthState.authenticated(user)),
    );
  }
}

02 · نبذة عني

أبني منتجات Mobile جاهزة للإنتاج باستخدام Flutter.

أنا Software Engineer وSenior Flutter Developer بخبرة أكثر من 5 سنوات في التطوير الاحترافي، مع تركيز أساسي على بناء وصيانة تطبيقات Mobile جاهزة للإنتاج. لدي خبرة عملية في تطوير أكثر من 15 تطبيقًا ومشروع Production، باستخدام Flutter وDart وClean Architecture وBLoC/Cubit وFirebase وREST APIs وWebSockets وPayment Gateways وSQL وGit وPerformance Optimization. كما منحتني خبرتي المبكرة في تطوير Web باستخدام ASP.NET/.NET وSQL أساسًا جيدًا في فهم Backend Systems وقواعد البيانات العلائقية وكيفية ارتباط تطبيقات Mobile بالأنظمة البرمجية الأخرى. أتعامل مع الهندسة البرمجية كعملية تعلّم وتطوّر مستمرة، وأستمتع بحل المشاكل التقنية المعقدة بقدر استمتاعي بتسليم منتج نظيف وموثوق.

"الكود النظيف ليس رفاهية شكلية، بل استراتيجية عمل تُحقّق المرونة وسرعة التطوير دون التضحية بالاستقرار."

كيف أعمل

أعمل على مختلف مراحل دورة حياة التطبيق، بدءًا من فهم المتطلبات وتصميم الـArchitecture وحتى التطوير وAPI Integration والاختبار والـDeployment والصيانة المستمرة. أعمل بشكل مريح سواء بشكل مستقل مع العملاء، حيث أترجم متطلبات العمل مباشرة إلى حلول تقنية، أو ضمن فريق بالتعاون الوثيق مع مطوّري الـbackend والمصممين وأصحاب المنتج.

ما أركّز عليه

أركّز على بناء تطبيقات ليست فقط عملية، بل أيضًا قابلة للصيانة والتوسع وذات أداء جيد وموثوقة وجاهزة لبيئات Production الحقيقية، دون إغفال تجربة استخدام سلسة ومدروسة.
